JS FETCH (servidor)
1. La espera: async / await
Pedir algo a un servidor tarda (milisegundos o segundos). JS no se queda congelado: fetch devuelve una promesa («te aviso cuando llegue»). Con await le dices «espera aquí hasta que llegue». await solo funciona dentro de una función async.

// Simulamos un servidor lento con una promesa que tarda 1 segundo
function servidorLento() {
  return new Promise(resolve => {
    setTimeout(() => resolve("✈️ datos listos"), 1000);  // responde en 1 s
  });
}

async function cargar() {          // async: aquí dentro se puede usar await
  console.log("1. Pido los datos...");
  const datos = await servidorLento();   // espera SIN congelar la página
  console.log("2. Llegaron:", datos);
}

cargar();
console.log("Mientras tanto, JS sigue con otras cosas");  // ¡sale antes del 2!

3. Un servidor de verdad (necesita internet)
JSONPlaceholder es una API gratuita de práctica. Pedimos sus usuarios y los convertimos en pasajeros.
async function cargarUsuarios() {
  try {
    const resp = await fetch("https://jsonplaceholder.typicode.com/users");

    // fetch NO lanza error con 404 o 500: hay que revisar resp.ok
    if (!resp.ok) {
      throw new Error("El servidor respondió " + resp.status);
    }

    const usuarios = await resp.json();          // array de 10 usuarios

    // map: convertimos cada usuario en un pasajero
    const pasajeros = usuarios.map((u, i) => ({
      nombre: u.name,
      asiento: `${i + 1}A`,
    }));
    pasajeros.slice(0, 5).forEach(p => console.log(p.asiento, p.nombre));

  } catch (error) {
    // entra aquí si no hay internet o si lanzamos el Error de arriba
    console.log("❌ No se pudo cargar:", error.message);
  }
}

cargarUsuarios();

async function registrar(pasajero) {
  try {
    const resp = await fetch("https://jsonplaceholder.typicode.com/posts", {
      method: "POST",                                   // tipo de petición
      headers: { "Content-Type": "application/json" },  // "te mando JSON"
      body: JSON.stringify(pasajero),                   // objeto → texto JSON
    });
    const guardado = await resp.json();                 // lo que devuelve el servidor
    console.log("Estado:", resp.status);                // 201 = creado
    console.log("Guardado con id:", guardado.id, guardado);
  } catch (error) {
    console.log("❌", error.message);
  }
}

    console.log(p.nombre, p.asiento);
  })
  .catch(err => {                     // si algo falló en cualquier paso
    console.log("❌", err.message);
  });


6. fetch + lista enlazada + DOM
Todo junto: el botón pide los pasajeros a la API, los mete en la lista enlazada y los dibuja.

<button id="cargar">🌐 Cargar pasajeros del servidor</button>

<script>
  // --- Lista enlazada mínima ---
  class Nodo {
    constructor(data) { this.data = data; this.next = null; }
  }
  class ListaEnlazada {
    constructor() { this.head = null; }
    insertarFinal(data) {
      const nuevo = new Nodo(data);
      if (!this.head) { this.head = nuevo; return; }   // lista vacía
      let actual = this.head;
      while (actual.next) actual = actual.next;        // camina al último
      actual.next = nuevo;                             // engancha
    }
  }
  const lista = new ListaEnlazada();

  // --- Dibujar recorriendo la lista ---
  function dibujar() {
    let html = "", actual = lista.head;
    while (actual) {
      html += `<span class="nodo">${actual.data.nombre}</span> →`;
      actual = actual.next;
    }
    document.getElementById("tren").innerHTML = html + " null";
  }

  // --- fetch al hacer clic ---
  document.getElementById("cargar").onclick = async () => {
    const estado = document.getElementById("estado");
    estado.textContent = "⏳ Cargando...";
    try {
      const resp = await fetch("https://jsonplaceholder.typicode.com/users");
      if (!resp.ok) throw new Error("Error " + resp.status);
      const usuarios = await resp.json();
      usuarios.forEach(u => lista.insertarFinal({ nombre: u.username }));
      estado.textContent = `✅ ${usuarios.length} pasajeros cargados`;
      dibujar();
    } catch (e) {
      estado.textContent = "❌ Sin conexión: " + e.message;
    }
  };
</script>

7. Conectar con TU backend (Flask)
Cuando tengas tu servidor, solo cambias la URL. Así se ve el código del lado del navegador:
const API = "/api/pasajeros";        // ruta de tu servidor Flask

// GET: traer todos los pasajeros guardados
async function obtenerPasajeros() {
  const resp = await fetch(API);
  if (!resp.ok) throw new Error("No se pudo obtener la lista");
  return await resp.json();          // ej: [{nombre:"Ana", asiento:"1A"}, ...]
}

// POST: guardar un pasajero nuevo
async function crearPasajero(pasajero) {
  const resp = await fetch(API, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(pasajero),  // { nombre, asiento }
  });
  return await resp.json();          // el servidor responde lo guardado
}

// DELETE: borrar por nombre (la ruta incluye el dato)
async function eliminarPasajero(nombre) {
  const resp = await fetch(`${API}/${encodeURIComponent(nombre)}`, {
    method: "DELETE",
  });
  return resp.ok;                    // true si se borró
}
⚠️ Errores típicos con fetch
Olvidar await → obtienes Promise {<pending>} en vez de los datos. · Olvidar .json() → tienes la respuesta, no los datos. · CORS → el servidor no permite que tu página le pida datos (se configura en el backend). · Abrir el .html con doble clic y pedir /api/... → no hay servidor detrás.
